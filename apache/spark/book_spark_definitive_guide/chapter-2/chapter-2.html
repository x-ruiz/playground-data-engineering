spark = SparkSession.builder.appName("Chapter2Examples").getOrCreate()
sc = spark.sparkContext
sc.setLogLevel("ERROR")
# Reading data is a lazy transformation. Spark peeks at a few rows only to infer the schema
# but waits to perform the operation until the take or show action
flightData2015 = spark \
    .read \
    .option("inferSchema", "true") \
    .option("header", "true") \
    .csv("flight-data/2015-summary.csv")

flightData2015.take(5)

# Turn any Dataframe into a table or a view to query with SQL
# there is no performance difference as both "compile" to the 
# same underlying plan.
flightData2015.createOrReplaceTempView("flight_data_2015")

# Compare plans of the sql method vs dataframe method
sqlWay = spark.sql("""
SELECT DEST_COUNTRY_NAME, count(1)
FROM flight_data_2015
GROUP BY DEST_COUNTRY_NAME
""")

dataFrameWay = flightData2015 \
    .groupBy("DEST_COUNTRY_NAME") \
    .count()

sqlWay.explain() # action to show plan
